# Path函数
给路径参数添加范围限制，注解。
```
import uvicorn
from fastapi import FastAPI,Path,Query
from pydantic import BaseModel

app = FastAPI()

@app.get("/book/{id}")
async def say_hello(id: int = Path(..., title="The ID of the book", ge=1, le=1000)):

    return {"message": f"这是第 {id} 本书"}

  

@app.get("/book/{name}")
async def say_hello(name: str = Path(..., title="The name of the book", min_length=1, max_length=10)):

    return {"message": f"这是书名 {name}"}
```

# 查询参数
可以使用Query函数对查询参数进行注解和范围限制
```
import uvicorn
from fastapi import FastAPI,Path,Query
from pydantic import BaseModel

app = FastAPI()

@app.get("/news/news_list")
async def news_list(

    skip: int = Query(0, description="Number of items to skip",lt=10),

    limit: int = Query(10, description="Number of items to limit")

    ):

    return {"message": f"这是新闻列表，跳过 {skip} 条，显示 {limit} 条"}
```

# 请求体参数
![](assets/04_参数/file-20260302135156977.png)
```
import uvicorn
from fastapi import FastAPI,Path,Query
from pydantic import BaseModel

app = FastAPI()

class User(BaseModel):
    username: str
    password: str


@app.post("/user/login")
async def login(user: User):
    return {"message": f"用户名：{user.username}，密码：{user.password}"}
```

# 请求体参数注解Field
```
class User(BaseModel):

    username: str = Field(..., description="The username of the user", min_length=3, max_length=100)

    password: str = Field(..., description="The password of the user", min_length=6, max_length=100)

  
  

@app.post("/user/login")

async def login(user: User):

    return {"message": f"用户名：{user.username}，密码：{user.password}"}

  

@app.post("/user/register")

async def register(user: User):

    return {"message": f"注册成功，用户名：{user.username}，密码：{user.password}"}
```
