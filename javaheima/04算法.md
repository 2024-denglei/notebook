# 快速排序
```java
package javabase.day03;  
  
public class Sort_ {  
    public static void main(String[] args) {  
        int[] arrys = {3,1,6,5,4,2,8,7,9};  
        quickly_sort(arrys,0,arrys.length-1);  
        for (int i = 0; i < arrys.length; i++) {  
            System.out.print(arrys[i] + " ");  
        }  
    }  
  
    // 首先初始化一个基准数,以及start,end指针  
    // 先判断end指针指向的数据是否比基准数大,如果大就--,如果小就停下  
    // 然后判断start指针指向的数据是否比基准数小,如果小就++,如果大就停下来  
    // 交换end和start对应的元素  
    //然后重复上面的流程,直到start和end都指向的同一个元素  
    // 然后交换基准数和该元素的位置.完成第一轮的排序  
    //  
    public static void quickly_sort(int[] arr,int i,int j){  
        // 递归出口  
        if (i >= j){  
            return;  
        }  
        // 初始化基准数和两个指针
        int base_number = arr[i];  
        int start = i;  
        int end = j;  
  
  
        while(start < end){  
            // 找到需要交换的end的位置  
            while(true){  
                if (arr[end] > base_number) {  
                    end--;  
                }else{  
                    break;  
                }  
            }  
            //找到需要交换的start的位置  
            while(true){  
                if(arr[start] < base_number){  
                    start++;  
                }else{  
                    break;  
                }  
            }  
            //交换end和start的值  
            int temp = arr[end];  
            arr[end] = arr[start];  
            arr[start] = temp;  
        }  
  
        // 当end=start打破循环后,交换base_number和它们指向的值,基准数归位 
        int temp1 = arr[end];  
        arr[end] = base_number;  
        base_number = temp1;  
  
        // 递归左右两侧  
        quickly_sort(arr, i, end - 1);  
        quickly_sort(arr, end + 1, j);  
  
  
    }  
}
```
